93: 	    private fun initSearch() {
94: 	        binding.search.setOnQueryTextListener(object : SearchView.OnQueryTextListener {
95: 	            override fun onQueryTextSubmit(query: String?): Boolean {
96: 	                if (query?.startsWith("!") == true)
97: 	                    requireContext().openUrl(Constants.URL_DUCK_SEARCH + query.replace(" ", "%20"))
98: 	                else if (adapter.itemCount == 0) // && requireContext().searchOnPlayStore(query?.trim()).not())
99: 	                    requireContext().openSearch(query?.trim())
100: 	                else
101: 	                    adapter.launchFirstInList()
102: 	                return true
103: 	            }
104: 	
105: 	            override fun onQueryTextChange(newText: String): Boolean {
106: 	                try {
107: 	                    adapter.filter.filter(newText)
108: 	                    binding.appDrawerTip.visibility = View.GONE
109: 	                    binding.appRename.visibility = if (canRename && newText.isNotBlank()) View.VISIBLE else View.GONE
110: 	                    binding.recyclerView.post { rebuildLetterPositions() }
111: 	                    return true
112: 	                } catch (e: Exception) {
113: 	                    e.printStackTrace()
114: 	                }
115: 	                return false
116: 	            }
117: 	        })
118: 	    }

120: 	    private fun initAdapter() {
121: 	        adapter = AppDrawerAdapter(

123: 	            prefs.appLabelAlignment,
124: 	            appClickListener = {
125: 	                if (it.appPackage.isEmpty())

137: 	                        findNavController().popBackStack(R.id.mainFragment, false)
138: 	                    else

142: 	            appInfoListener = {
143: 	                openAppInfo(
144: 	                    requireContext(),
145: 	                    it.user,
146: 	                    it.appPackage
147: 	                )
148: 	                findNavController().popBackStack(R.id.mainFragment, false)
149: 	            },
150: 	            appDeleteListener = {
151: 	                requireContext().apply {
152: 	                    if (isSystemApp(it.appPackage))

154: 	                    else

174: 	                if (prefs.firstHide) {

189: 	        linearLayoutManager = object : LinearLayoutManager(requireContext()) {
190: 	            override fun scrollVerticallyBy(

215: 	    private fun initLetterIndex() {
216: 	        // Populate A–Z index vertically
217: 	        binding.letterIndex.removeAllViews()
218: 	        letters.forEach { ch ->
219: 	            val tv = layoutInflater.inflate(R.layout.item_letter_index, binding.letterIndex, false) as TextView
220: 	            tv.text = ch.toString()
221: 	            binding.letterIndex.addView(tv)
222: 	        }
223: 	
224: 	        // Touch handler maps Y to letter, scrolls and shows overlay
225: 	        binding.letterIndex.setOnTouchListener { v, event ->
226: 	            when (event.action) {
227: 	                MotionEvent.ACTION_DOWN, MotionEvent.ACTION_MOVE -> {
228: 	                    val letter = letterForY(v.height, event.y)
229: 	                    if (letter != null) {
230: 	                        showLetterOverlay(letter)
231: 	                        scrollToLetter(letter)
232: 	                    }
233: 	                    true
234: 	                }
235: 	                MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
236: 	                    overlayHideHandler.removeCallbacks(overlayHideRunnable)
237: 	                    overlayHideHandler.postDelayed(overlayHideRunnable, 600)
238: 	                    true
239: 	                }
240: 	                else -> false
241: 	            }
242: 	        }
243: 	    }
244: 	
245: 	    private fun letterForY(viewHeight: Int, y: Float): Char? {
246: 	        if (viewHeight <= 0) return null
247: 	        val slot = (y / viewHeight * letters.size).toInt().coerceIn(0, letters.size - 1)
248: 	        return letters[slot]
249: 	    }

251: 	    private fun showLetterOverlay(letter: Char) {
252: 	        binding.letterOverlay.text = letter.toString()
253: 	        binding.letterOverlay.visibility = View.VISIBLE
254: 	        overlayHideHandler.removeCallbacks(overlayHideRunnable)
255: 	    }

257: 	    private fun rebuildLetterPositions() {
258: 	        letterPositions.clear()
259: 	        val list = adapter.appFilteredList
260: 	        for (i in list.indices) {
261: 	            val ch = normalizedFirstChar(list[i].appLabel)
262: 	            if (ch != null && ch in 'A'..'Z' && ch !in letterPositions) {
263: 	                letterPositions[ch] = i
264: 	            }
265: 	        }
266: 	    }
267: 	
268: 	    private fun normalizedFirstChar(label: String): Char? {
269: 	        val trimmed = label.trim()
270: 	        if (trimmed.isEmpty()) return null
271: 	        val base = Normalizer.normalize(trimmed, Normalizer.Form.NFD)
272: 	            .replace(Regex("\\p{InCombiningDiacriticalMarks}+"), "")
273: 	            .trim()
274: 	        return base.firstOrNull { it.isLetterOrDigit() }?.uppercaseChar()
275: 	    }
276: 	
277: 	    private fun scrollToLetter(letter: Char) {
278: 	        letterPositions[letter]?.let { pos ->
279: 	            linearLayoutManager.scrollToPositionWithOffset(pos, 0)
280: 	        }
281: 	    }

283: 	    private fun showDelayStepper(packageName: String) {
284: 	        try {
285: 	            var value = prefs.getPerAppDelaySeconds(packageName) ?: (prefs.mindfulDelayMs / 1000)
286: 	            val dlg = AlertDialog.Builder(requireContext())
287: 	                .setTitle(getString(R.string.mindful_delay))
288: 	                .setMessage(packageName)
289: 	                .setPositiveButton(R.string.okay) { _, _ ->
290: 	                    prefs.setPerAppDelaySeconds(packageName, value)
291: 	                }
292: 	                .setNegativeButton(R.string.close, null)
293: 	                .setView(LayoutInflater.from(requireContext()).inflate(R.layout.view_delay_stepper, null))
294: 	                .create()
295: 	            dlg.show()
296: 	            val tvVal = dlg.findViewById<TextView>(R.id.tvValue)
297: 	            val minus = dlg.findViewById<View>(R.id.btnMinus)
298: 	            val plus = dlg.findViewById<View>(R.id.btnPlus)
299: 	            tvVal?.text = value.toString()
300: 	            minus?.setOnClickListener { value = (value - 1).coerceIn(0, 10); tvVal?.text = value.toString() }
301: 	            plus?.setOnClickListener { value = (value + 1).coerceIn(0, 10); tvVal?.text = value.toString() }
302: 	        } catch (e: Exception) {
303: 	            e.printStackTrace()
304: 	        }
305: 	    }

310: 	                binding.appDrawerTip.visibility = View.VISIBLE
311: 	                binding.appDrawerTip.isSelected = true
312: 	            }
313: 	        }
314: 	        if (flag == Constants.FLAG_HIDDEN_APPS) {
315: 	            viewModel.hiddenApps.observe(viewLifecycleOwner) {
316: 	                it?.let {
317: 	                    adapter.setAppList(it.toMutableList())
318: 	                }
319: 	            }
320: 	        } else {
321: 	            viewModel.appList.observe(viewLifecycleOwner) {
322: 	                it?.let { appModels ->
323: 	                    adapter.setAppList(appModels.toMutableList())
324: 	                    adapter.filter.filter(binding.search.query)
325: 	                    binding.recyclerView.post { rebuildLetterPositions() }
326: 	                }
327: 	            }
328: 	        }
329: 	    }

358: 	    private fun getRecyclerViewOnScrollListener(): RecyclerView.OnScrollListener {
359: 	        return object : RecyclerView.OnScrollListener() {
360: 	
361: 	            var onTop = false
362: 	
363: 	            override fun onScrollStateChanged(recyclerView: RecyclerView, newState: Int) {
364: 	                super.onScrollStateChanged(recyclerView, newState)
365: 	                when (newState) {
366: 	
367: 	                    RecyclerView.SCROLL_STATE_DRAGGING -> {
368: 	                        onTop = !recyclerView.canScrollVertically(-1)
369: 	                        if (onTop)
370: 	                            binding.search.hideKeyboard()
371: 	                    }
372: 	
373: 	                    RecyclerView.SCROLL_STATE_IDLE -> {
374: 	                        if (!recyclerView.canScrollVertically(1))
375: 	                            binding.search.hideKeyboard()
376: 	                        else if (!recyclerView.canScrollVertically(-1))
377: 	                            if (!onTop && isRemoving.not())
378: 	                                binding.search.showKeyboard(prefs.autoShowKeyboard)
379: 	                    }
380: 	                }
381: 	            }
382: 	        }
383: 	    }

388: 	            viewModel.checkForMessages.call()
389: 	    }
